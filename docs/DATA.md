# DealTrace data model and mock data

Seed files live in src/mocks/data/*.json, generated by scripts/generate-seed.ts (TypeScript, run with tsx, seeded PRNG mulberry32 with seed 42, so the output is identical every run). Types live in src/types and match these interfaces exactly.

## Interfaces
```ts
type Severity = 'critical' | 'high' | 'medium' | 'low';
type FindingStatus = 'Open' | 'In review' | 'Resolved' | 'Dismissed';
type ProductCode = 'GAP' | 'VSC' | 'TIRE_WHEEL' | 'PAINT_FABRIC' | 'PREPAID_MAINT' | 'THEFT_ETCH';

interface User { id: string; name: string; role: string; email: string; initials: string } // email: name@harborline.example
interface Dealer { id: string; name: string; city: string; state: 'CA'; group: 'Harborline Auto Group' }
interface Product { code: ProductCode; name: string; provider_id: string; price: number; term_months: number; max_miles?: number }
interface Vehicle { year: number; make: string; model: string; trim: string; vin: string; condition: 'new'|'used';
  msrp: number; factory_warranty_months: number; factory_warranty_miles: number } // VIN 17 chars, no I/O/Q
interface DealVersion { version: number; created_at: string; actor_id: string; reason: string | null;
  vehicle_price: number; down_payment: number; trade_in_value: number; trade_in_payoff: number;
  amount_financed: number; apr: number; term_months: number; monthly_payment: number; total_of_payments: number;
  products: Product[]; label: string }
interface Deal { id: string; dealer_id: string; customer_display: string; vehicle: Vehicle; lender_id: string;
  deal_date: string; status: 'In review'|'Ready to fund'|'Funded'|'On hold'; fi_manager_id: string; salesperson_id: string;
  versions: DealVersion[]; current_version: number; risk_score: number; refund_exposure: number;
  customer_exposure: number; open_findings: number }
type EventType = 'QUOTE_CREATED'|'PRICE_AGREED'|'CREDIT_APPROVED'|'PRODUCT_ADDED'|'PRODUCT_REMOVED'|'RATE_CHANGED'
  |'TERM_CHANGED'|'PRICE_CHANGED'|'CONTRACT_SIGNED'|'DECISION_RECORDED';
interface DealEvent { id: string; deal_id: string; seq: number; type: EventType; field: string | null;
  old_value: string | null; new_value: string | null; actor_id: string; reason: string | null;
  timestamp: string; version_after: number; hash: string; prev_hash: string }
type FindingType = 'LATE_ADD_ON'|'PAYMENT_MASKING'|'RATE_AFTER_APPROVAL'|'WARRANTY_OVERLAP'|'LOW_BENEFIT_GAP'
  |'DUPLICATE_COVERAGE'|'MISSING_REASON'|'PRICE_OUTLIER'|'RECORDKEEPING_GAP';
interface Finding { id: string; deal_id: string; type: FindingType; title: string; severity: Severity; status: FindingStatus;
  detected_at: string; version_from: number; version_to: number; rule_id?: string;
  evidence: { label: string; value: string; event_id?: string }[]; uncertainty: number; // 0 to 1
  refund_exposure: number; customer_exposure: number; assignee_id: string;
  label: 'Potential violation – human review required' }
interface BenefitCheck { deal_id: string; product_code: ProductCode;
  verdict: 'Likely benefit'|'Partial benefit'|'Little or no benefit'; score: number; reasons: string[];
  inputs: { ltv: number; negative_equity: number; warranty_overlap_months: number; existing_coverage: string } }
interface Rule { id: string; name: string; description: string; jurisdiction: string; source: string; category: string;
  effective_from: string; effective_to?: string; recorded_at: string; version: string; severity: Severity;
  status: 'Active'|'Draft'; enabled: boolean }
interface BlastRadius { rule_id: string; change_description: string; affected_deals: number; affected_dealers: number;
  estimated_refund_exposure: number; sample_deal_ids: string[] }
interface AgentSummary { deal_id: string; summary: string; key_points: string[]; suggested_actions: string[];
  tools_used: string[]; generated_at: string }
interface Decision { finding_id: string; action: 'approve'|'fix_before_funding'|'escalate'|'dismiss'; note: string;
  actor_id: string; timestamp: string }
interface AuditEvent { id: string; timestamp: string; actor_id: string; action: string; entity: string;
  entity_id: string; hash: string; prev_hash: string }
interface PricingPoint { deal_id: string; dealer_id: string; product_code: ProductCode; price: number; segment: string; is_outlier: boolean }
interface Accuracy { dataset: string; overall: { precision: number; recall: number; f1: number };
  per_detector: { detector: string; precision: number; recall: number; f1: number; support: number }[];
  confusion_matrix: { tp: number; fp: number; fn: number; tn: number } }
```

## Hash chain
hash = SHA-256 hex of (prev_hash + canonical JSON of the record without its `hash` field). Canonical JSON = keys sorted alphabetically at every level, no whitespace. The first record uses prev_hash of 64 zeros. The same function runs in the seed script (node crypto) and in the browser verifier (Web Crypto), so verification really recomputes the chain.

## Rules for realistic data
- No round numbers in amounts, scores or counts (use $31,300.00, 87, 0.874).
- Deal dates between Oct 1 and Nov 12, 2026; timestamps in business hours, Pacific time.
- Real vehicle makes and models (Toyota RAV4, Honda CR-V, Ford F-150, Tesla Model Y, Hyundai Tucson). APR 4.9% to 13.9%; terms 48, 60, 72 or 84.
- About 1.57 add-ons per deal on average; GAP on about 40% of deals, service contracts on about 45%.
- About 60% of deals clean, 25% with medium findings, 15% with high findings.
- Every payment and total computed with src/lib/finance.ts.
